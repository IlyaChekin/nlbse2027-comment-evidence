## Воспроизведение

Команды запускаются из корня экспериментов на Banana. Полные файлы индекса и замороженные матрицы остаются на вычислительном хосте. Сводки, хеши, прогнозы и параметры моделей сохранены рядом с этим отчётом.

```bash
export NLBSE_SCANNER_WORKERS=6 OMP_NUM_THREADS=2 OPENBLAS_NUM_THREADS=2
export NLBSE_NODE="$PWD/nlbse/round19/weak_satd/runtime/node"
p=.venv/bin/python
r=nlbse/round25
for part in test_a test_b; do
  $p "$r/length_eval.py" "$part"
  $p "$r/length_sparse.py" "$part"
  $p "$r/neighbors_eval.py" "$part"
  $p "$r/skip_sparse.py" "$part"
  $p "$r/view_scores.py" "$part"
done
for stage in length_eval length_sparse skip_sparse view_scores; do
  $p nlbse/cuda_launch.py "$r/$stage.py" official
done
$p nlbse/cuda_launch.py "$r/view_public.py"
$p "$r/length_audit.py"
$p "$r/view_audit.py"
$p "$r/build_release.py"
$p nlbse/round12/verify_release.py nlbse/submission/NLBSE2027_round25_sections_tool.zip \
  nlbse/data/raw/test-00000-of-00001.parquet "$r/release_verification" \
  --entry "$r/predict_sections.py" --cuda-launch
python3 "$r/summarize.py"
```
